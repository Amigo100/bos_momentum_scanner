def get_highlight_threshold(entry_date: str) -> float:
    """
    Return appropriate highlight threshold based on position age.
    
    Weekly timeframe positions need time to develop - newer positions
    get lower thresholds.

        return 5.0    # Early momentum
    elif days_held <= 30:
        return 10.0   # Building position
    elif days_held <= 60:
        return 15.0   # Standard threshold
    else:

def get_position_age_category(entry_date: str) -> str:
    """Categorize position by age for appropriate handling."""

    if days_held <= 14:
        return 'early'      # For early_movers category
    elif days_held <= 60:
        return 'developing' # Standard handling
    else:
        return 'mature'     # Should be performing

def format_holding_period(entry_date: str) -> str:
    """Format holding period for display. REQUIRED on all P&L."""
    days_held = (datetime.now() - datetime.strptime(entry_date, '%Y-%m-%d')).days
    
    if days_held == 0:
        return "today"
    elif days_held == 1:
        return "1 day"
    elif days_held < 7:
        return f"{days_held} days"
    else:
        weeks = days_held // 7
        if weeks == 1:
            return "1 week"
        else:
            return f"{weeks} weeks"

```python
# signal_tracker.py

import json
import pandas as pd
import yfinance as yf
from datetime import datetime, timedelta
from config import (
    CELEBRATION_TIERS, 
    get_highlight_threshold, 
    format_holding_period,
    TICKER_LIMITS,
    THRESHOLDS,
)

class SignalTracker:
    def __init__(self, portfolio_path: str, celebrations_path: str):
        self.portfolio_path = portfolio_path
        self.celebrations_path = celebrations_path
        self.portfolio = self._load_portfolio()
        self.celebrations = self._load_celebrations()
    
    def _load_portfolio(self) -> pd.DataFrame:
        """Load portfolio.csv."""
        return pd.read_csv(self.portfolio_path)
    
    def _load_celebrations(self) -> dict:
        """Load celebrations.json tracking."""
        try:
            with open(self.celebrations_path) as f:
                return json.load(f)
        except FileNotFoundError:
            return {}
    
    def _save_celebrations(self):
        """Save celebrations.json."""
        with open(self.celebrations_path, 'w') as f:
            json.dump(self.celebrations, f, indent=2)
    
    def get_open_positions(self) -> list:
        """Get all OPEN positions with current prices."""
        open_pos = self.portfolio[self.portfolio['status'] == 'OPEN']
        positions = []
        
        for _, row in open_pos.iterrows():
            ticker = row['ticker']
            try:
                current_price = yf.Ticker(ticker).info.get('regularMarketPrice')
                if current_price:
                    entry_price = row['entry_price']
                    pnl_pct = ((current_price / entry_price) - 1) * 100
                    days_held = (datetime.now() - pd.to_datetime(row['entry_date'])).days
                    
                    positions.append({
                        'ticker': ticker,
                        'entry_date': row['entry_date'],
                        'entry_price': entry_price,
                        'current_price': current_price,
                        'pnl_pct': round(pnl_pct, 2),
                        'days_held': days_held,
                        'weeks_held': days_held // 7,
                        'holding_period': format_holding_period(row['entry_date']),  # 🆕 MANDATORY
                        'theme': row['theme'],
                        'signal_type': row['signal_type'],
                    })
            except Exception as e:
                print(f"Error fetching {ticker}: {e}")
        
        return positions
    
    def filter_public_positions(self, positions: list) -> list:
        """
        🆕 CRITICAL SAFEGUARD: Remove ALL positions that shouldn't be shown publicly.
        
        This function MUST be called before ANY public output.
        """
        return [
            p for p in positions
            if p['pnl_pct'] > THRESHOLDS['max_loss_to_mention']  # Must be positive
            and p.get('status') != 'STOPPED'  # Never show stopped
        ]
    
    def get_recent_closes(self, days: int = 14) -> list:
        """Get positions closed in last N days - WINNERS ONLY."""
        cutoff = datetime.now() - timedelta(days=days)
        closed = self.portfolio[
            (self.portfolio['status'].isin(['CLOSED', 'STOPPED'])) &
            (pd.to_datetime(self.portfolio['exit_date']) >= cutoff)
        ]
        
        results = []
        for _, row in closed.iterrows():
            pnl_pct = ((row['exit_price'] / row['entry_price']) - 1) * 100
            
            # 🆕 CRITICAL: Only include winners for public content
            if pnl_pct > 0 and row['status'] != 'STOPPED':
                days_held = (pd.to_datetime(row['exit_date']) - pd.to_datetime(row['entry_date'])).days
                results.append({
                    'ticker': row['ticker'],
                    'entry_price': row['entry_price'],
                    'exit_price': row['exit_price'],
                    'pnl_pct': round(pnl_pct, 2),
                    'days_held': days_held,
                    'weeks_held': days_held // 7,
                    'holding_period': f"{days_held // 7} weeks" if days_held >= 7 else f"{days_held} days",
                    'theme': row['theme'],
                    'status': row['status'],
                })
        
        return results
    
    def detect_milestone_crossings(self, positions: list) -> list:
        """Find positions that crossed celebration thresholds."""
        # Filter to public positions first
        public_positions = self.filter_public_positions(positions)
        crossings = []
        
        for pos in public_positions:
            ticker = pos['ticker']
            pnl = pos['pnl_pct']
            
            # Initialize tracking if needed
            if ticker not in self.celebrations:
                self.celebrations[ticker] = {
                    '25_pct': None,
                    '50_pct': None,
                    '100_pct': None,
                }
            
            # Check each threshold
            for tier_name, tier_config in CELEBRATION_TIERS.items():
                threshold = tier_config['threshold']
                key = f"{int(threshold)}_pct"
                
                if pnl >= threshold and self.celebrations[ticker].get(key) is None:
                    # New crossing!
                    self.celebrations[ticker][key] = datetime.now().strftime('%Y-%m-%d')
                    crossings.append({
                        **pos,
                        'milestone': tier_name,
                        'threshold': threshold,
                        'emoji': tier_config['emoji'],
                        'headline': tier_config['headline'],
                    })
        
        # Save updated celebrations
        self._save_celebrations()
        
        return crossings
    
    def get_early_movers(self, max_age_days: int = 14, min_gain: float = 5.0) -> list:
        """Get new signals showing early strength - WINNERS ONLY."""
        positions = self.get_open_positions()
        public_positions = self.filter_public_positions(positions)  # 🆕 Apply filter
        
        return [
            p for p in public_positions
            if p['days_held'] <= max_age_days and p['pnl_pct'] >= min_gain
        ]
    
    def calculate_spy_comparison(self, positions: list) -> dict:
        """
        Calculate fair SPY comparison using matched holding periods.
        
        🆕 Must use same holding period for each position vs SPY.
        """
        # Filter to public positions first
        public_positions = self.filter_public_positions(positions)
        
        if not public_positions:
            return {'valid': False, 'reason': 'No public positions'}
        
        spy = yf.Ticker('SPY')
        comparisons = []
        
        for pos in public_positions:
            try:
                entry_date = pos['entry_date']
                spy_hist = spy.history(start=entry_date)
                
                if len(spy_hist) > 1:
                    spy_return = ((spy_hist['Close'].iloc[-1] / spy_hist['Close'].iloc[0]) - 1) * 100
                    alpha = pos['pnl_pct'] - spy_return
                    
                    comparisons.append({
                        'ticker': pos['ticker'],
                        'pos_return': pos['pnl_pct'],
                        'spy_return': round(spy_return, 2),
                        'alpha': round(alpha, 2),
                        'days_held': pos['days_held'],
                        'holding_period': pos['holding_period'],
                    })
            except Exception:
                continue
        
        if not comparisons:
            return {'valid': False, 'reason': 'No valid comparisons'}
        
        avg_pos = sum(c['pos_return'] for c in comparisons) / len(comparisons)
        avg_spy = sum(c['spy_return'] for c in comparisons) / len(comparisons)
        avg_alpha = avg_pos - avg_spy
        avg_days = sum(c['days_held'] for c in comparisons) / len(comparisons)
        
        return {
            'valid': True,
            'avg_position_return': round(avg_pos, 2),
            'avg_spy_return': round(avg_spy, 2),
            'avg_alpha': round(avg_alpha, 2),
            'avg_days_held': int(avg_days),
            'avg_holding_period': f"{int(avg_days // 7)} weeks" if avg_days >= 7 else f"{int(avg_days)} days",
            'position_count': len(comparisons),
            'details': comparisons,
        }
```

def validate_tweet_before_queue(tweet: dict, existing_tweets: list) -> tuple[bool, list]:
    """
    🆕 Run all validations before adding tweet to queue.
    
    Returns (is_valid, list_of_errors)
    """
    errors = []
    text = tweet.get('text', '')
    category = tweet.get('category', '')
    
    # 1. Check banned terms
    banned = check_banned_terms(text)
    if banned:
        errors.append(f"Banned terms found: {banned}")
    
    # 2. Check killed categories
    if category in KILLED_CATEGORIES:
        errors.append(f"Category '{category}' is killed")
    
    # 3. Check tweet length
    if not validate_tweet_length(text):
        errors.append("Tweet exceeds 280 characters")
    
    # 4. Check TEAL branding for signal tweets
    if category in ['buy_signal', 'thread_buy_signal', 'milestone_alerts']:
        if not validate_teal_branding(text, category):
            errors.append("Missing TEAL branding in signal tweet")
    
    # 5. Check holding period for P&L tweets
    has_pnl = '%' in text and ('+' in text or '-' in text)
    if has_pnl and not validate_holding_period_present(text, True):
        errors.append("P&L shown without holding period")
    
    # 6. Check for negative P&L (losers)
    import re
    negative_pnl = re.findall(r'-\d+\.?\d*%', text)
    if negative_pnl:
        errors.append(f"Negative P&L found (losers shown): {negative_pnl}")
    
    # 7. Check ticker frequency
    tickers = re.findall(r'\$([A-Z]+)', text)
    for ticker in tickers:
        if not check_ticker_frequency(ticker, existing_tweets):
            errors.append(f"Ticker ${ticker} mentioned too many times this week")
    
    # 8. Check for US-specific content
    us_terms = ['roth', '401k', 'pdt rule', 'pattern day']
    for term in us_terms:
        if term in text.lower():
            errors.append(f"US-specific term found: {term}")
    
    return (len(errors) == 0, errors)

def validate_before_posting(tweet: dict) -> bool:
    """
    🆕 Final validation before posting to Twitter.
    
    This is the LAST LINE OF DEFENSE.
    """
    text = tweet.get('text', '')
    
    # 1. Check for losing positions
    import re
    if re.search(r'-\d+\.?\d*%', text):
        print(f"🚨 BLOCKED: Negative P&L in tweet: {text[:50]}...")
        return False
    
    # 2. Check for banned terms
    for term in BANNED_TERMS:
        if term.lower() in text.lower():
            print(f"🚨 BLOCKED: Banned term '{term}' in tweet")
            return False
    
    # 3. Check for US-specific content
    us_terms = ['roth', '401k', 'pdt']
    for term in us_terms:
        if term in text.lower():
            print(f"🚨 BLOCKED: US-specific term '{term}' in tweet")
            return False
    
    # 4. Check length
    if not validate_tweet_length(text):
        print(f"🚨 BLOCKED: Tweet exceeds 280 chars")
        return False
    
    return True
